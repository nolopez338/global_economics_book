	\section*{1. Community Water Pricing Support}
		
	A city council surveys \textbf{220 households} about support for a new water pricing plan. In the sample, \textbf{143 households} say they support the plan.
	
	Construct and interpret a \textbf{95\% confidence interval} for the population proportion of households that support the plan.
	
	\subsubsection*{(C7) Constructs the confidence interval for a proportion}
	To construct the confidence interval for a proportion, we identify \(x\) and \(n\) and compute \(\hat{p}\), check normal approximation conditions \(n\hat{p} \ge 10\) and \(n(1-\hat{p}) \ge 10\), choose the correct \(z^*\) for the confidence level, compute \(SE\) and the margin of error, construct and report the confidence interval, and interpret the interval in context.
	
	\textbf{Step 1: Sample proportion.}
	\[
	\hat{p} = \frac{x}{n} = \frac{143}{220} = 0.65
	\]
	
	\textbf{Step 2: Large-sample conditions.}
	\[
	n\hat{p} = 220(0.65) = 143 \ge 10, \qquad n(1-\hat{p}) = 220(0.35) = 77 \ge 10
	\]
	
	\textbf{Step 3: Standard error.}
	\[
	SE = \sqrt{\frac{\hat{p}(1-\hat{p})}{n}} = \sqrt{\frac{0.65(0.35)}{220}} \approx 0.0322
	\]
	
	\textbf{Step 4: Margin of error (95\% confidence, \(z^* = 1.96\)).}
	\[
	E = z^* \cdot SE = 1.96(0.0322) \approx 0.0631
	\]
	
	\textbf{Step 5: Confidence interval.}
	\[
	\hat{p} \pm E = 0.65 \pm 0.0631 \Rightarrow [0.587,\ 0.713]
	\]
	
	\textbf{Interpretation.}
	We are 95\% confident that between about 58.7\% and 71.3\% of households in the city support the water pricing plan.

	\[
	n\hat{p} = 180(0.25) = 45 \ge 10, \qquad n(1-\hat{p}) = 180(0.75) = 135 \ge 10
	\]
	
	\textbf{Step 3: Standard error.}
	\[
	SE = \sqrt{\frac{\hat{p}(1-\hat{p})}{n}} = \sqrt{\frac{0.25(0.75)}{180}} \approx 0.0323

	\section*{3. Small Business Grant Renewals}
	
	A development agency tracks \textbf{360 grants} issued to small businesses. During the renewal period, \textbf{54 grants} were not renewed.
	
	Construct and interpret \textbf{90\% and 95\% confidence intervals} for the population proportion of grants that are not renewed.
	
	\subsubsection*{(C7) Constructs the confidence interval for a proportion}
	To construct the confidence interval for a proportion, we identify \(x\) and \(n\) and compute \(\hat{p}\), check normal approximation conditions \(n\hat{p} \ge 10\) and \(n(1-\hat{p}) \ge 10\), choose the correct \(z^*\) for the confidence level, compute \(SE\) and the margin of error, construct and report the confidence interval, and interpret the interval in context.
	
	\textbf{Step 1: Sample proportion.}
	\[
	\hat{p} = \frac{x}{n} = \frac{54}{360} = 0.15
	\]
	
	\textbf{Step 2: Large-sample conditions.}
	\[
	n\hat{p} = 360(0.15) = 54 \ge 10, \qquad n(1-\hat{p}) = 360(0.85) = 306 \ge 10
	\]
	
	\textbf{Step 3: Standard error.}
	\[
	SE = \sqrt{\frac{\hat{p}(1-\hat{p})}{n}} = \sqrt{\frac{0.15(0.85)}{360}} \approx 0.0188
	\]
	
	\textbf{Step 4: Margin of error and confidence interval for 90\% confidence.}
	\[
	E_{90} = z^* \cdot SE = 1.645(0.0188) \approx 0.0309

	\textbf{Interpretation.}
	We are 90\% confident that between about 11.9\% and 18.1\% of grants are not renewed, and we are 95\% confident that between about 11.3\% and 18.7\% are not renewed.

	\section*{5. Port Inspection Compliance}
	
	A port authority evaluates compliance with a new inspection checklist. Sample A includes \textbf{250 inspections} with \textbf{170} compliant. Sample B includes \textbf{180 inspections} with \textbf{110} compliant. Assume both samples are drawn from the same population of inspections.
	
	Construct and interpret a \textbf{95\% confidence interval} for the population proportion of compliant inspections for each sample.
	
	\subsubsection*{(C7) Constructs the confidence interval for a proportion}
	To construct the confidence interval for a proportion, we identify \(x\) and \(n\) and compute \(\hat{p}\), check normal approximation conditions \(n\hat{p} \ge 10\) and \(n(1-\hat{p}) \ge 10\), choose the correct \(z^*\) for the confidence level, compute \(SE\) and the margin of error, construct and report the confidence interval, and interpret the interval in context.
	
	\textbf{Sample A}
	
	\textbf{Step 1: Sample proportion.}
	\[
	\hat{p}_A = \frac{x}{n} = \frac{170}{250} = 0.68
	\]
	
	\textbf{Step 2: Large-sample conditions.}
	\[
	n\hat{p}_A = 250(0.68) = 170 \ge 10, \qquad n(1-\hat{p}_A) = 250(0.32) = 80 \ge 10
	\]
	
	\textbf{Step 3: Standard error.}
	\[
	SE_A = \sqrt{\frac{\hat{p}_A(1-\hat{p}_A)}{n}} = \sqrt{\frac{0.68(0.32)}{250}} \approx 0.0295
	\]
	
	\textbf{Step 4: Margin of error (95\% confidence, \(z^* = 1.96\)).}
	\[
	E_A = z^* \cdot SE_A = 1.96(0.0295) \approx 0.0578
	\]
	
	\textbf{Step 5: Confidence interval.}
	\[
	\hat{p}_A \pm E_A = 0.68 \pm 0.0578 \Rightarrow [0.622,\ 0.738]
	\]
	
	\textbf{Sample B}
	
	\textbf{Step 1: Sample proportion.}
	\[
	\hat{p}_B = \frac{x}{n} = \frac{110}{180} \approx 0.611
	\]
	
	\textbf{Step 2: Large-sample conditions.}
	\[
	n\hat{p}_B = 180(0.611) \approx 110 \ge 10, \qquad n(1-\hat{p}_B) = 180(0.389) \approx 70 \ge 10
	\]
	
	\textbf{Step 3: Standard error.}
	\[
	SE_B = \sqrt{\frac{\hat{p}_B(1-\hat{p}_B)}{n}} = \sqrt{\frac{0.611(0.389)}{180}} \approx 0.0363
	\]
	
	\textbf{Step 4: Margin of error (95\% confidence, \(z^* = 1.96\)).}
	\[
	E_B = z^* \cdot SE_B = 1.96(0.0363) \approx 0.0711
	\]
	
	\textbf{Step 5: Confidence interval.}
	\[
	\hat{p}_B \pm E_B \approx 0.611 \pm 0.0711 \Rightarrow [0.540,\ 0.682]
	\]
	
	\textbf{Interpretation.}
	We are 95\% confident that the compliance rate is between about 62.2\% and 73.8\% for Sample A, and between about 54.0\% and 68.2\% for Sample B.
	
	\section*{6. Public Market Vendor Licensing}
	
	A city reviews vendor licensing in two districts. Sample A includes \textbf{300 vendors} with \textbf{96} licensed. Sample B includes \textbf{210 vendors} with \textbf{60} licensed.
	
	Construct and interpret \textbf{90\% and 98\% confidence intervals} for the population proportion of licensed vendors for each sample.
	
	\subsubsection*{(C7) Constructs the confidence interval for a proportion}
	To construct the confidence interval for a proportion, we identify \(x\) and \(n\) and compute \(\hat{p}\), check normal approximation conditions \(n\hat{p} \ge 10\) and \(n(1-\hat{p}) \ge 10\), choose the correct \(z^*\) for the confidence level, compute \(SE\) and the margin of error, construct and report the confidence interval, and interpret the interval in context.
	
	\textbf{Sample A}
	
	\textbf{Step 1: Sample proportion.}
	\[
	\hat{p}_A = \frac{x}{n} = \frac{96}{300} = 0.32
	\]
	
	\textbf{Step 2: Large-sample conditions.}
	\[
	n\hat{p}_A = 300(0.32) = 96 \ge 10, \qquad n(1-\hat{p}_A) = 300(0.68) = 204 \ge 10
	\]
	
	\textbf{Step 3: Standard error.}
	\[
	SE_A = \sqrt{\frac{\hat{p}_A(1-\hat{p}_A)}{n}} = \sqrt{\frac{0.32(0.68)}{300}} \approx 0.0269

	\textbf{Step 4: Margin of error and confidence interval for 90\% confidence.}
	\[
	E_{A,90} = z^* \cdot SE_A = 1.645(0.0269) \approx 0.0442

	\[
	\hat{p}_A \pm E_{A,90} = 0.32 \pm 0.0442 \Rightarrow [0.276,\ 0.364]
	\]
	
	\textbf{Step 5: Margin of error and confidence interval for 98\% confidence.}
	\[
	E_{A,98} = z^* \cdot SE_A = 2.326(0.0269) \approx 0.0626

	\[
	E_{B,98} = z^* \cdot SE_B = 2.326(0.0312) \approx 0.0727

	\[
	\hat{p}_B \pm E_{B,98} \approx 0.286 \pm 0.0727 \Rightarrow [0.213,\ 0.358]
	\]
	
	\textbf{Interpretation.}
	We are 90\% confident that between about 27.6\% and 36.4\% of vendors are licensed in Sample A and between about 23.4\% and 33.7\% are licensed in Sample B. We are 98\% confident that between about 25.7\% and 38.3\% of vendors are licensed in Sample A and between about 21.3\% and 35.8\% are licensed in Sample B.
