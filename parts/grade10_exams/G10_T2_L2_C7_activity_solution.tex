		\subsection*{Problem 1 — Pop-up Cafe Launch}
		
		\textbf{Problem.}
		A business considers opening a weekend pop-up cafe. Profit depends on the state of nature.
		
		\[
		\begin{array}{lcc}
			\hline
			& S_1\,(0.6) & S_2\,(0.4) \\
			\hline
			\text{Launch} & 15 & -5 \\
			\hline
		\end{array}
		\]
		
		Determine whether the cafe should launch using Expected Value.
		
		\textbf{Solution.}
		Expected value:
		\[
		EV = 15\cdot 0.6 + (-5)\cdot 0.4
		\]
		Compute:
		\[
		EV = 9 - 2 = 7
		\]
		Since \(EV>0\), the cafe should launch.

		\subsection*{Problem 3 — Two Advertising Packages}
		
		\textbf{Problem.}
		A firm chooses between two advertising packages. Outcomes depend on the state of nature.
		
		\[
		\begin{array}{lcc}
			\hline
			& S_1\,(0.7) & S_2\,(0.3) \\
			\hline
			A & 20 & 4 \\
			B & 14 & 8 \\
			\hline
		\end{array}
		\]
		
		Which package is better using Expected Value.
		
		\textbf{Solution.}
		Compute expected value of A:
		\[
		EV(A)=20\cdot 0.7 + 4\cdot 0.3
		\]
		Compute:
		\[
		EV(A)=14 + 1.2 = 15.2
		\]
		Compute expected value of B:
		\[
		EV(B)=14\cdot 0.7 + 8\cdot 0.3
		\]
		Compute:
		\[
		EV(B)=9.8 + 2.4 = 12.2
		\]
		Since \(EV(A)>EV(B)\), package A is preferred.

			\hline
		\end{array}

		\textbf{Solution.}
		Plan A:
		\[
		EV(A)=28\cdot 0.65 + 2\cdot 0.35
		\]
		Compute:
		\[
		EV(A)=18.2 + 0.7 = 18.9
		\]
		Plan B:
		\[
		EV(B)=20\cdot 0.65 + 10\cdot 0.35
		\]
		Compute:
		\[
		EV(B)=13 + 3.5 = 16.5
		\]
		Plan C:
		\[
		EV(C)=16\cdot 0.65 + 12\cdot 0.35
		\]
		Compute:
		\[
		EV(C)=10.4 + 4.2 = 14.6
		\]
		Since \(EV(A)\) is largest, plan A is optimal.

		\subsection*{Problem 7 — Three Distribution Options with Uncertain Demand}
		
		\textbf{Problem.}
		A firm must choose among three distribution options. Payoffs depend on demand.
		
		\[
		\begin{array}{lcc}
			\hline
			& S_1\,(p) & S_2\,(1-p) \\
			\hline
			A & 32 & -2 \\
			B & 22 & 10 \\
			C & 16 & 14 \\
			\hline
		\end{array}
		\]
		
		Let \(p\) be the probability of state \(S_1\).  
		Determine for which values of \(p\) each option is optimal.
		
		\textbf{Solution.}
		Compute expected value of each option.
		
		Option A:
		\[
		EV(A)=32\cdot p + (-2)\cdot(1-p)
		\]
		Expand:
		\[
		EV(A)=32p - 2 + 2p = 34p - 2
		\]
		
		Option B:
		\[
		EV(B)=22\cdot p + 10\cdot(1-p)
		\]
		Expand:
		\[
		EV(B)=22p + 10 - 10p = 12p + 10
		\]
		
		Option C:
		\[
		EV(C)=16\cdot p + 14\cdot(1-p)
		\]
		Expand:
		\[
		EV(C)=16p + 14 - 14p = 2p + 14
		\]
		
		Now compare pairwise using \(>0\) conditions.
		
		A better than B:
		\[
		EV(A)-EV(B)>0 \Rightarrow (34p-2)-(12p+10)>0
		\]
		Simplify:
		\[
		34p-2-12p-10>0 \Rightarrow 22p-12>0 \Rightarrow p>\frac{12}{22}=\frac{6}{11}
		\]
		
		B better than C:
		\[
		EV(B)-EV(C)>0 \Rightarrow (12p+10)-(2p+14)>0
		\]
		Simplify:
		\[
		12p+10-2p-14>0 \Rightarrow 10p-4>0 \Rightarrow p>\frac{4}{10}=\frac{2}{5}
		\]
		
		A better than C:
		\[
		EV(A)-EV(C)>0 \Rightarrow (34p-2)-(2p+14)>0
		\]
		Simplify:
		\[
		34p-2-2p-14>0 \Rightarrow 32p-16>0 \Rightarrow p>\frac{16}{32}=\frac{1}{2}
		\]
		
		Determine optimal option by intervals:
		\begin{itemize}
			\item If \(p<\frac{2}{5}\), then \(EV(C)>EV(B)\). Also \(EV(C)>EV(A)\) because \(EV(A)-EV(C)>0\) requires \(p>\frac{1}{2}\). So for \(p<\frac{2}{5}\), option C is optimal.
			\item If \(\frac{2}{5}<p<\frac{6}{11}\), then \(EV(B)>EV(C)\) and \(EV(A)<EV(B)\) (since \(p<\frac{6}{11}\)). So in this interval, option B is optimal.
			\item If \(p>\frac{6}{11}\), then \(EV(A)>EV(B)\) and also \(EV(A)>EV(C)\) (since \(\frac{6}{11}>\frac{1}{2}\)). So for \(p>\frac{6}{11}\), option A is optimal.
		\end{itemize}

		\subsection*{Problem 8 — Mobile App Launch}
		
		\textbf{Problem.}
		A developer considers launching a mobile app. Profit depends on adoption and platform stability.
		
		\[
		\begin{array}{lcc}
			\hline
			& \text{Stable}\,(0.7) & \text{Unstable}\,(0.3) \\
			\hline
			\text{High adoption}\,(0.6) & 30 & 8 \\
			\text{Low adoption}\,(0.4) & -6 & -14 \\
			\hline
		\end{array}
		\]
		
		Assume independence between adoption and stability. Determine whether launching is profitable.
		
		\textbf{Solution.}
		By independence, joint probabilities:
		\[
		P(\text{High,Stable})=0.6\cdot 0.7,\quad
		P(\text{High,Unstable})=0.6\cdot 0.3,\quad
		P(\text{Low,Stable})=0.4\cdot 0.7,\quad
		P(\text{Low,Unstable})=0.4\cdot 0.3
		\]
		Expected value:
		\[
		EV=30(0.6\cdot 0.7)+8\bigl(0.6\cdot 0.3\bigr)+(-6)\bigl(0.4\cdot 0.7\bigr)+(-14)\bigl(0.4\cdot 0.3\bigr)
		\]
		Compute:
		\[
		EV=30(0.42)+8(0.18)-6(0.28)-14(0.12)
		\]
		\[
		EV=12.6+1.44-1.68-1.68=10.68
		\]
		Since \(EV>0\), launching is profitable.

		\subsection*{Problem 9 — Sponsorship Strategy Choice}
		
		\textbf{Problem.}
		A team must choose between two sponsorship strategies. Outcomes depend on audience interest
		and campaign execution.
		
		\[
		\begin{array}{lcc}
			\hline
			\text{State of nature} & A & B \\
			\hline
			\text{High interest}\,(0.5),\,\text{Effective}\,(0.6) & 26 & 20 \\
			\text{High interest}\,(0.5),\,\text{Ineffective}\,(0.4) & 12 & 16 \\
			\text{Low interest}\,(0.5),\,\text{Effective}\,(0.6) & -4 & 4 \\
			\text{Low interest}\,(0.5),\,\text{Ineffective}\,(0.4) & -10 & -6 \\
			\hline
		\end{array}
		\]
		
		Assume interest and execution are independent. Which strategy is better using Expected Value.
		
		\textbf{Solution.}
		Compute \(EV_A\) by weighting each payoff:
		\[
		EV_A=26(0.5\cdot 0.6)+12\bigl(0.5\cdot 0.4\bigr)+(-4)\bigl(0.5\cdot 0.6\bigr)+(-10)\bigl(0.5\cdot 0.4\bigr)
		\]
		Compute:
		\[
		EV_A=26(0.3)+12(0.2)-4(0.3)-10(0.2)
		\]
		\[
		EV_A=7.8+2.4-1.2-2=7.0
		\]
		Compute \(EV_B\):
		\[
		EV_B=20(0.5\cdot 0.6)+16\bigl(0.5\cdot 0.4\bigr)+4\bigl(0.5\cdot 0.6\bigr)+(-6)\bigl(0.5\cdot 0.4\bigr)
		\]
		Compute:
		\[
		EV_B=20(0.3)+16(0.2)+4(0.3)-6(0.2)
		\]
		\[
		EV_B=6+3.2+1.2-1.2=9.2
		\]
		Since \(EV_B>EV_A\), strategy B is preferred.

		\subsection*{Problem 10 — Portfolio Selection with Two Economic Factors}
		
		\textbf{Problem.}
		An investor must choose between two portfolios. Outcomes depend on economic growth
		and interest rates.
		
		\[
		\begin{array}{lcc}
			\hline
			\text{State of nature} & X & Y \\
			\hline
			\text{Strong growth}\,(0.4),\,\text{Low rates}\,(0.7) & 24 & 18 \\
			\text{Strong growth}\,(0.4),\,\text{High rates}\,(0.3) & 10 & 14 \\
			\text{Weak growth}\,(0.6),\,\text{Low rates}\,(0.7) & 4 & 8 \\
			\text{Weak growth}\,(0.6),\,\text{High rates}\,(0.3) & -8 & -2 \\
			\hline
		\end{array}
		\]
		
		Assume growth and rates are independent. Which portfolio is better using Expected Value.
		
		\textbf{Solution.}
		Compute \(EV_X\):
		\[
		EV_X=24(0.4\cdot 0.7)+10\bigl(0.4\cdot 0.3\bigr)+4\bigl(0.6\cdot 0.7\bigr)+(-8)\bigl(0.6\cdot 0.3\bigr)
		\]
		Compute:
		\[
		EV_X=24(0.28)+10(0.12)+4(0.42)-8(0.18)
		\]
		\[
		EV_X=6.72+1.2+1.68-1.44=8.16
		\]
		Compute \(EV_Y\):
		\[
		EV_Y=18(0.4\cdot 0.7)+14\bigl(0.4\cdot 0.3\bigr)+8\bigl(0.6\cdot 0.7\bigr)+(-2)\bigl(0.6\cdot 0.3\bigr)
		\]
		Compute:
		\[
		EV_Y=18(0.28)+14(0.12)+8(0.42)-2(0.18)
		\]
		\[
		EV_Y=5.04+1.68+3.36-0.36=9.72
		\]
		Since \(EV_Y>EV_X\), portfolio Y is preferred.
