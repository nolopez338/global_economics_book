	\ExamSection{C10: Full end-to-end decision analysis}
	\begin{ExamProblems}
		\item
		\subsection*{Problem description}
		A food-processing company must choose one packaging strategy for next quarter:
		$P$ (premium line), $Q$ (standard line), or $R$ (hybrid line).
		Demand can be high ($0.60$) or low ($0.40$).
		Operating cost can be low ($0.50$), medium ($0.30$), or high ($0.20$).
		Revenues (thousand USD):
		\begin{itemize}
			\item High demand: $P=500,\ Q=540,\ R=520$.
			\item Low demand: $P=320,\ Q=300,\ R=310$.
		\end{itemize}
		Costs (thousand USD):
		\begin{itemize}
			\item $P$: low $180$, medium $220$, high $260$.
			\item $Q$: low $210$, medium $250$, high $295$.
			\item $R$: low $195$, medium $235$, high $275$.
		\end{itemize}
		Build the payoff table, apply EV, Maximin, and Minimax Regret, produce ordered rankings, then use a staged decision tree to choose a final strategy.

		\subsection*{C10}
		Step 1. Build the payoff table from narrative data.
		Joint state probabilities:
		\[
		\begin{aligned}
		&\Pr(H,L)=0.60\cdot 0.50=0.30,\quad \Pr(H,M)=0.60\cdot 0.30=0.18,\quad \Pr(H,Hc)=0.60\cdot 0.20=0.12,\\
		&\Pr(L,L)=0.40\cdot 0.50=0.20,\quad \Pr(L,M)=0.40\cdot 0.30=0.12,\quad \Pr(L,Hc)=0.40\cdot 0.20=0.08.
		\end{aligned}
		\]
		Payoff = revenue $-$ cost.
		\begin{center}
			\begin{tabular}{l c c c c}
				\toprule
				State of nature & Probability & $P$ & $Q$ & $R$ \\
				\midrule
				High demand, low cost & 0.30 & $500-180=320$ & $540-210=330$ & $520-195=325$ \\
				High demand, medium cost & 0.18 & $500-220=280$ & $540-250=290$ & $520-235=285$ \\
				High demand, high cost & 0.12 & $500-260=240$ & $540-295=245$ & $520-275=245$ \\
				Low demand, low cost & 0.20 & $320-180=140$ & $300-210=90$ & $310-195=115$ \\
				Low demand, medium cost & 0.12 & $320-220=100$ & $300-250=50$ & $310-235=75$ \\
				Low demand, high cost & 0.08 & $320-260=60$ & $300-295=5$ & $310-275=35$ \\
				\bottomrule
			\end{tabular}
		\end{center}

		Step 2. Bayes criterion (EV).
		\[
		\begin{aligned}
		EV(P)&=0.30(320)+0.18(280)+0.12(240)+0.20(140)+0.12(100)+0.08(60)\\
		&=96+50.4+28.8+28+12+4.8=220.0,\\
		EV(Q)&=0.30(330)+0.18(290)+0.12(245)+0.20(90)+0.12(50)+0.08(5)\\
		&=99+52.2+29.4+18+6+0.4=205.0,\\
		EV(R)&=0.30(325)+0.18(285)+0.12(245)+0.20(115)+0.12(75)+0.08(35)\\
		&=97.5+51.3+29.4+23+9+2.8=213.0.
		\end{aligned}
		\]
		EV ranking:
		\[
		P\,(220.0)>R\,(213.0)>Q\,(205.0).
		\]

		Step 3. Maximin criterion.
		\[
		\min(P)=60,\quad \min(Q)=5,\quad \min(R)=35.
		\]
		Maximin ranking:
		\[
		P\,(60)>R\,(35)>Q\,(5).
		\]

		Step 4. Minimax Regret criterion.
		State best payoffs:
		\[
		\max=\{330,290,245,140,100,60\}.
		\]
		Regret table:
		\begin{center}
			\begin{tabular}{l c c c c c c c}
				\toprule
				Alternative & $HL$ & $HM$ & $HHc$ & $LL$ & $LM$ & $LHc$ & Max regret \\
				\midrule
				$P$ & $330-320=10$ & $290-280=10$ & $245-240=5$ & $140-140=0$ & $100-100=0$ & $60-60=0$ & 10 \\
				$Q$ & $330-330=0$ & $290-290=0$ & $245-245=0$ & $140-90=50$ & $100-50=50$ & $60-5=55$ & 55 \\
				$R$ & $330-325=5$ & $290-285=5$ & $245-245=0$ & $140-115=25$ & $100-75=25$ & $60-35=25$ & 25 \\
				\bottomrule
			\end{tabular}
		\end{center}
		Minimax Regret ranking (lower is better):
		\[
		P\,(10)>R\,(25)>Q\,(55).
		\]

		Step 5. Staged decision tree and final choice.
		Filtering rule: keep best 2 EV options, then choose smaller Minimax Regret.
		\begin{center}
			\begin{tabular}{l}
				\toprule
				Decision tree \\
				\midrule
				$D_0$: $\{P,Q,R\}$ \\
				$\rightarrow D_1$ (EV filter): keep $\{P,R\}$ \\
				$\rightarrow D_2$ (Minimax Regret): compare $P(10)$ vs $R(25)$ \\
				Leaves: $P$, $R$ \\
				Final leaf selected: $P$ \\
				\bottomrule
			\end{tabular}
		\end{center}
		Final explanation: $P$ is first in EV and Maximin, and also has the smallest maximum regret, so the tree process consistently ends at $P$.

		\newpage
		\item
		\subsection*{Problem description}
		An online retailer must choose one pricing strategy among six options $S_1,S_2,S_3,S_4,S_5,S_6$.
		Demand states are optimistic ($0.45$), regular ($0.35$), and weak ($0.20$).
		Payoffs (thousand USD):
		\begin{center}
			\begin{tabular}{l c c c c c c c}
				\toprule
				State & Probability & $S_1$ & $S_2$ & $S_3$ & $S_4$ & $S_5$ & $S_6$ \\
				\midrule
				Optimistic & 0.45 & 150 & 140 & 170 & 130 & 160 & 145 \\
				Regular & 0.35 & 90 & 100 & 80 & 110 & 95 & 105 \\
				Weak & 0.20 & 20 & 45 & -10 & 60 & 35 & 25 \\
				\bottomrule
			\end{tabular}
		\end{center}
		Apply EV, Maximin, and Minimax Regret step by step, generate full ordered rankings, and then apply this staged rule:
		keep best 4 EV options $\rightarrow$ keep best 3 Maximin options $\rightarrow$ choose smallest Minimax Regret.

		\subsection*{C10}
		Step 1. EV calculations.
		\[
		\begin{aligned}
		EV(S_1)&=0.45(150)+0.35(90)+0.20(20)=103.00,\\
		EV(S_2)&=0.45(140)+0.35(100)+0.20(45)=107.00,\\
		EV(S_3)&=0.45(170)+0.35(80)+0.20(-10)=102.50,\\
		EV(S_4)&=0.45(130)+0.35(110)+0.20(60)=109.00,\\
		EV(S_5)&=0.45(160)+0.35(95)+0.20(35)=112.25,\\
		EV(S_6)&=0.45(145)+0.35(105)+0.20(25)=107.00.
		\end{aligned}
		\]
		EV ranking:
		\[
		S_5\,(112.25)>S_4\,(109.00)>S_2\,(107.00)=S_6\,(107.00)>S_1\,(103.00)>S_3\,(102.50).
		\]
		There is a tie between $S_2$ and $S_6$.

		Step 2. Maximin calculations.
		\[
		\begin{aligned}
		\min(S_1)&=20,\ \min(S_2)=45,\ \min(S_3)=-10,\\
		\min(S_4)&=60,\ \min(S_5)=35,\ \min(S_6)=25.
		\end{aligned}
		\]
		Maximin ranking:
		\[
		S_4\,(60)>S_2\,(45)>S_5\,(35)>S_6\,(25)>S_1\,(20)>S_3\,(-10).
		\]

		Step 3. Minimax Regret calculations.
		Best state payoffs:
		\[
		\max(\text{Optimistic})=170,\quad \max(\text{Regular})=110,\quad \max(\text{Weak})=60.
		\]
		Regret table:
		\begin{center}
			\begin{tabular}{l c c c c}
				\toprule
				Alternative & Optimistic & Regular & Weak & Max regret \\
				\midrule
				$S_1$ & $170-150=20$ & $110-90=20$ & $60-20=40$ & 40 \\
				$S_2$ & $170-140=30$ & $110-100=10$ & $60-45=15$ & 30 \\
				$S_3$ & $170-170=0$ & $110-80=30$ & $60-(-10)=70$ & 70 \\
				$S_4$ & $170-130=40$ & $110-110=0$ & $60-60=0$ & 40 \\
				$S_5$ & $170-160=10$ & $110-95=15$ & $60-35=25$ & 25 \\
				$S_6$ & $170-145=25$ & $110-105=5$ & $60-25=35$ & 35 \\
				\bottomrule
			\end{tabular}
		\end{center}
		Minimax Regret ranking (lower is better):
		\[
		S_5\,(25)>S_2\,(30)>S_6\,(35)>S_1\,(40)=S_4\,(40)>S_3\,(70).
		\]
		There is a tie between $S_1$ and $S_4$.

		Step 4. Decision tree for the staged rule.
		\begin{itemize}
			\item EV top 4: $\{S_5,S_4,S_2,S_6\}$.
			\item On this subset, Maximin top 3: $S_4(60),S_2(45),S_5(35)$, so keep $\{S_4,S_2,S_5\}$.
			\item On this subset, Minimax Regret values are $S_5(25),S_2(30),S_4(40)$.
		\end{itemize}
		\begin{center}
			\begin{tabular}{l}
				\toprule
				Decision tree \\
				\midrule
				$D_0$: $\{S_1,S_2,S_3,S_4,S_5,S_6\}$ \\
				$\rightarrow D_1$ (EV filter): $\{S_5,S_4,S_2,S_6\}$ \\
				$\rightarrow D_2$ (Maximin filter): $\{S_4,S_2,S_5\}$ \\
				$\rightarrow D_3$ (Minimax Regret choice): compare $S_4(40),S_2(30),S_5(25)$ \\
				Leaves: $S_4$, $S_2$, $S_5$ \\
				Final leaf selected: $S_5$ \\
				\bottomrule
			\end{tabular}
		\end{center}

		Step 5. Final conclusion.
		The final decision is $S_5$ because it is first in EV, third but acceptable in Maximin after filtering, and best in Minimax Regret among finalists.
